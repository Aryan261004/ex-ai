def call_model_function(
    images,
    call_model_args=None,
    expected_keys=None
):

    target_class = call_model_args["class_idx"]

    images = tf.convert_to_tensor(images, dtype=tf.float32)

    with tf.GradientTape() as tape:

        tape.watch(images)

        processed = tf.keras.applications.inception_v3.preprocess_input(
            images
        )

        predictions = model(processed, training=False)

        score = predictions[:, target_class]

    gradients = tape.gradient(score, images)

    return {
        saliency.base.INPUT_OUTPUT_GRADIENTS:
        gradients.numpy()
    }
