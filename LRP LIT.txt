def forward(x, W1, b1, W2, b2):

    hidden = np.maximum(0, x @ W1 + b1)

    output = hidden @ W2 + b2

    return hidden, output

hidden, logit = forward(x, W1, b1, W2, b2)

def lrp(x, W1, W2, b1, b2, epsilon=0):

    hidden, output = forward(x, W1, b1, W2, b2)

    # Output relevance
    R_output = output

    # Output to hidden layer
    z2 = hidden * W2

    denominator2 = np.sum(z2)

    if abs(denominator2) < 1e-12:
        R_hidden = np.zeros_like(hidden)
    else:
        R_hidden = (z2 / denominator2) * R_output

    # Hidden layer to input words
    z1 = x[:, None] * W1

    denominator1 = np.sum(z1, axis=0)

    denominator1 = denominator1 + epsilon * np.where(
        denominator1 >= 0, 1, -1
    )

    R_input = np.divide(
        z1,
        denominator1[None, :],
        out=np.zeros_like(z1),
        where=np.abs(denominator1[None, :]) > 1e-12
    ) @ R_hidden

    return R_input


-----------


def grad_x_input(x, W1, W2, b1):

    hidden = np.maximum(0, x @ W1 + b1)

    gradient = W1 @ (W2 * (hidden > 0))

    attribution = x * gradient

    return attribution


------------

# Set biases to zero

# LRP on bias-free model
